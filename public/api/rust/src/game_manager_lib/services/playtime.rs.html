1//! Rastreamento local de tempo jogado, usado como fallback para plataformas
2//! que não expõem playtime via API própria (todas exceto Steam/Itch/Indiegala).
3//!
4//! Funciona observando, por polling, se o executável resolvido do jogo (já
5//! persistido em `games.executable_path` na importação) está em execução.
6//! Não depende do processo ser filho direto do Playlite — cobre também os
7//! casos em que o Playlite só abriu o launcher da plataforma (protocolo ou
8//! executável do launcher) e o jogo de fato sobe depois, por fora.
9
10use std::collections::HashSet;
11use std::path::PathBuf;
12use std::sync::Mutex;
13use std::time::Duration;
14
15use sysinfo::{ProcessRefreshKind, ProcessesToUpdate, RefreshKind, System};
16use tauri::{AppHandle, Manager};
17
18use crate::database::AppState;
19use crate::models::Platform;
20use crate::utils::status_logic;
21
22/// Mesma granularidade das plataformas com API oficial (Steam/Itch/Indiegala
23/// reportam playtime em minutos) — evita precisão inconsistente entre fontes.
24const POLL_INTERVAL: Duration = Duration::from_secs(60);
25
26/// Tempo máximo esperando o processo do jogo aparecer depois que o launcher
27/// foi aberto (ex: usuário demora navegando antes de clicar "Play", ou desiste).
28/// Evita task órfã rodando indefinidamente.
29const MAX_WAIT_FOR_START: Duration = Duration::from_secs(10 * 60);
30
31/// Evita observar o mesmo jogo duas vezes (ex: usuário clica "Jogar" repetidamente).
32#[derive(Default)]
33pub struct PlaytimeRegistry {
34    watching: Mutex<HashSet<String>>,
35}
36
37impl PlaytimeRegistry {
38    /// Retorna `true` se conseguiu reservar o watch (ninguém mais observando este jogo).
39    fn try_start(&self, game_id: &str) -> bool {
40        self.watching.lock().unwrap().insert(game_id.to_string())
41    }
42
43    fn finish(&self, game_id: &str) {
44        self.watching.lock().unwrap().remove(game_id);
45    }
46}
47
48/// Plataformas que já expõem playtime oficial — o tracker local nunca roda pra elas,
49/// pra não gerar dado duplicado/conflitante com o valor sincronizado na importação.
50pub fn has_official_playtime_source(platform: &Platform) -> bool {
51    matches!(
52        platform,
53        Platform::Steam | Platform::Itch | Platform::Indiegala
54    )
55}

57/// Dispara o watcher em background para o jogo informado. Idempotente: se já
58/// houver um watch em andamento para o mesmo `game_id`, não faz nada.
59pub fn watch_game(app: AppHandle, game_id: String, exe_path: PathBuf) {
60    let state = app.state::<AppState>();
61    if !state.playtime_registry.try_start(&game_id) {
62        return;
63    }
64
65    tauri::async_runtime::spawn(async move {
66        let mut sys = System::new_with_specifics(
67            RefreshKind::nothing().with_processes(ProcessRefreshKind::everything()),
68        );
69
70        let mut found = false;
71        let mut waited = Duration::ZERO;
72
73        loop {
74            tokio::time::sleep(POLL_INTERVAL).await;
75            sys.refresh_processes(ProcessesToUpdate::All, true);
76
77            let running = sys
78                .processes()
79                .values()
80                .any(|p| p.exe().map(|p| p == exe_path.as_path()).unwrap_or(false));
81
82            if running {
83                found = true;
84                if let Err(e) = increment_playtime(&app, &game_id) {
85                    tracing::warn!("[playtime] falha ao gravar incremento para {game_id}: {e}");
86                }
87            } else if found {
88                break; // processo existiu e sumiu -> sessão encerrada
89            } else {
90                waited += POLL_INTERVAL;
91                if waited >= MAX_WAIT_FOR_START {
92                    break; // desistiu de esperar o jogo abrir
93                }
94            }
95        }
96
97        app.state::<AppState>().playtime_registry.finish(&game_id);
98    });
99}

101/// Incrementa 1 minuto de playtime e recalcula `status`, no mesmo padrão usado
102/// pelos importadores (ex: `persist_indiegala_games`), pra manter a UI de
103/// status ("jogando"/"backlog") coerente sem esperar uma reimportação.
104fn increment_playtime(app: &AppHandle, game_id: &str) -> Result<(), rusqlite::Error> {
105    let state = app.state::<AppState>();
106    let conn = state.games_db.lock().unwrap();
107
108    let current: Option<i32> = conn.query_row(
109        "SELECT playtime FROM games WHERE id = ?1",
110        [game_id],
111        |row| row.get(0),
112    )?;
113    let new_playtime = current.unwrap_or(0) + 1;
114    let status = status_logic::calculate_status(new_playtime);
115
116    conn.execute(
117        "UPDATE games SET playtime = ?1, status = ?2, playtime_source = 'local' WHERE id = ?3",
118        rusqlite::params![new_playtime, status, game_id],
119    )?;
120    Ok(())
121}
