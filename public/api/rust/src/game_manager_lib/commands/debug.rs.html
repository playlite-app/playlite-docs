1use crate::database;
2use crate::errors::AppError;
3use tauri::{AppHandle, Manager};
4
5/// DEBUG — apenas para testes manuais via devtools.
6/// Busca o catálogo do Nexus Mods e popula a tabela `nexus_games` do cache,
7/// sem passar pelo fluxo de enriquecimento. Remover depois dos testes.
8#[tauri::command]
9pub async fn debug_populate_nexus_cache(app: AppHandle) -> Result<String, AppError> {
10    let api_key = database::get_secret(&app, "nexus_api_key")?;
11    if api_key.is_empty() {
12        return Err(AppError::ValidationError(
13            "Nexus API Key não configurada.".to_string(),
14        ));
15    }
16
17    let games = crate::services::integration::nexus::fetch_nexus_games(&api_key)
18        .await
19        .map_err(|e| AppError::ValidationError(format!("Erro ao buscar dados do Nexus: {}", e)))?;
20
21    let count = games.len();
22
23    let state: tauri::State<database::AppState> = app.state();
24    let cache_conn = state
25        .cache_db
26        .lock()
27        .map_err(|_| AppError::ValidationError("Falha ao travar cache_db".to_string()))?;
28
29    crate::services::cache::save_nexus_games_cache(&cache_conn, &games)
30        .map_err(|e| AppError::ValidationError(format!("Erro ao salvar cache: {}", e)))?;
31
32    Ok(format!("{} jogos do Nexus salvos no cache local.", count))
33}
