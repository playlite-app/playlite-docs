13/// Type alias para dados de backup
14pub type BackupDataTuple = (
15    Vec<Game>,
16    Vec<GameDetails>,
17    Vec<WishlistGame>,
18    Vec<GameExtras>,
19    Vec<SystemRequirements>,
20    Vec<GameDataPath>,
21    u32,
22);
23
24/// Função auxiliar interna para buscar dados do backup com transação ACID
25pub fn fetch_backup_data(state: &State<AppState>) -> Result<BackupDataTuple, AppError> {
26    let conn = state.games_db.lock()?;
27
28    // Inicia transação READ para consistência
29    conn.execute("BEGIN TRANSACTION", [])?;
30
31    let games = fetch_games(&conn)?;
32    let game_details = fetch_game_details(&conn)?;
33    let wishlist_game = fetch_wishlist(&conn)?;
34    let game_extras = fetch_game_extras(&conn)?;
35    let system_requirements = fetch_system_requirements(&conn)?;
36    let game_data_paths = fetch_game_data_paths(&conn)?;
37    let schema_version = current_schema_version(&conn)?;
38
39    conn.execute("COMMIT", [])?;
40
41    Ok((
42        games,
43        game_details,
44        wishlist_game,
45        game_extras,
46        system_requirements,
47        game_data_paths,
48        schema_version,
49    ))
50}

52/// Busca todos os jogos na biblioteca
53fn fetch_games(conn: &Connection) -> Result<Vec<Game>, AppError> {
54    let mut stmt = conn.prepare(
55        "SELECT id, name, cover_url, platform, platform_game_id, installed, import_confidence, install_path, executable_path, launch_args, user_rating, favorite, status, playtime, last_played, added_at FROM games"
56    )?;
57
58    let game_iter = stmt.query_map([], |row| {
59        Ok(Game {
60            id: row.get(0)?,
61            name: row.get(1)?,
62            cover_url: row.get(2)?,
63            genres: None,
64            developer: None,
65            platform: row.get::<_, String>(3)?.parse().unwrap_or(Platform::Outra),
66            platform_game_id: row.get(4)?,
67            installed: row.get(5)?,
68            import_confidence: row
69                .get::<_, Option<String>>(6)?
70                .and_then(|s| s.parse().ok()),
71            install_path: row.get(7)?,
72            executable_path: row.get(8)?,
73            launch_args: row.get(9)?,
74            user_rating: row.get(10)?,
75            favorite: row.get(11)?,
76            status: row.get(12)?,
77            playtime: row.get(13)?,
78            last_played: row.get(14)?,
79            added_at: row.get(15)?,
80            is_adult: false,
81        })
82    })?;
83
84    Ok(game_iter.collect::<Result<Vec<_>, _>>()?)
85}

87/// Busca todos os detalhes dos jogos na biblioteca
88fn fetch_game_details(conn: &Connection) -> Result<Vec<GameDetails>, AppError> {
89    let mut stmt = conn.prepare(
90        "SELECT
91        game_id, steam_app_id, developer, publisher, release_date, genres, tags, series,
92        description_raw, description_ptbr, background_image, critic_score,
93        steam_review_label, steam_review_count, steam_review_score, steam_review_updated_at,
94        esrb_rating, is_adult, adult_tags, external_links, median_playtime,
95        estimated_playtime
96     FROM game_details",
97    )?;
98
99    let details_iter = stmt.query_map([], |row| {
100        let links_json: Option<String> = row.get(19)?;
101        let external_links = links_json.and_then(|s| serde_json::from_str(&s).ok());
102
103        let tags_json: Option<String> = row.get(6)?;
104        let tags = tags_json.map(|s| crate::database::deserialize_tags(&s));
105
106        Ok(GameDetails {
107            game_id: row.get(0)?,
108            steam_app_id: row.get(1)?,
109            developer: row.get(2)?,
110            publisher: row.get(3)?,
111            release_date: row.get(4)?,
112            genres: row.get(5)?,
113            tags,
114            series: row.get(7)?,
115            description_raw: row.get(8)?,
116            description_ptbr: row.get(9)?,
117            background_image: row.get(10)?,
118            critic_score: row.get(11)?,
119            steam_review_label: row.get(12)?,
120            steam_review_count: row.get(13)?,
121            steam_review_score: row.get(14)?,
122            steam_review_updated_at: row.get(15)?,
123            esrb_rating: row.get(16)?,
124            is_adult: row.get(17).unwrap_or(false),
125            adult_tags: row.get(18)?,
126            external_links,
127            median_playtime: row.get(20)?,
128            estimated_playtime: row.get(21)?,
129        })
130    })?;
131
132    Ok(details_iter.collect::<Result<Vec<_>, _>>()?)
133}

163fn fetch_game_extras(conn: &Connection) -> Result<Vec<GameExtras>, AppError> {
164    let mut stmt = conn.prepare(
165        "SELECT
166            steam_app_id, pcgw_page_id, pcgw_page_name, engine,
167            available_on,
168            dx_versions, vulkan_versions, opengl_versions,
169            win64, linux64, macos_arm, macos_intel64,
170            ray_tracing, upscaling, frame_gen,
171            ultrawidescreen, four_k_support, hdr, high_fps, fov, borderless_windowed, color_blind,
172            controller_support, full_controller, playstation_controllers, xinput_controllers,
173            surround_sound, subtitles, closed_captions,
174            has_save_data, has_config_data,
175            languages_interface, languages_audio, languages_subtitles,
176            fetched_at
177         FROM game_extras
178         WHERE fetched_at IS NOT NULL",
179    )?;
180
181    let parse_json_vec = |s: Option<String>| -> Option<Vec<String>> {
182        s.and_then(|v| serde_json::from_str(&v).ok())

184
185    let iter = stmt.query_map([], |row| {
186        Ok(GameExtras {

188            pcgw_page_id: row.get(1)?,
189            pcgw_page_name: row.get(2)?,
